class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int left=0;
        int right=matrix.size()-1;
        int row=-1;

        while(left<=right){

            int m=(left+right)/2;
            if(matrix[m][0]<=target&&matrix[m][matrix[m].size()-1]>=target){
                row=m;
                left=right+1;
            }
            else if(matrix[m][0]>target){
                right=m-1;
            }
            else{
                left=m+1;
            }

        
        }
        if(row==-1){
            return false;
        }


        left=0;
        right=matrix[0].size()-1;

        while(left<=right){

            int m=(left+right)/2;
            if(matrix[row][m]==target){
                cout<<row<<" "<<m<<" ";
                return true;
            }
            else if(matrix[row][m]>target){
                right=m-1;
            }
            else{
                left=m+1;
            }

        
        }
        return false;
        
    }
};
