class Solution {
public:
    int findMin(vector<int> &nums) {
        int left=0;
        int right=nums.size()-1;
        
        while(left<right){
            int median=(left+right)/2;

            if(nums[median]<nums[right]){
                right=median;
            }
            else{
                left=median+1;
            }
        }

        return nums[left];

        
    }
};
