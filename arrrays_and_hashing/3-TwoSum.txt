class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {

        map<int,int>indices;

        for(int i=0;i <nums.size();i++){
            int current_value=nums[i];
            int necesary_value=target-current_value;
        
            if(indices.count(necesary_value)==0)
            {
                indices[current_value]=i;
            }
            else
            {
                vector<int>result={indices[necesary_value],i};
                return result;
            }
        }

        
    } 
    
};


note: i used a map to track the indices of each passed value
