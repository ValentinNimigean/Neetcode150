class Solution {
public:
    int maxArea(vector<int>& heights) {

        int left=0;
        int right=heights.size()-1;
        int maximum=0;

        while(left<right){

            int current=(right-left)*min(heights[right],heights[left]);
            if(current>maximum){
                maximum=current;
            }

            if(heights[left]<=heights[right])
            left++;
            else
            right--;

        }
     return maximum;   
    }
};
